import "dotenv/config";
import express from "express";
import path from "node:path";
import { fileURLToPath } from "node:url";
import crypto from "node:crypto";

const app = express();
const __dirname = path.dirname(fileURLToPath(import.meta.url));

app.use(express.json({ limit: "32kb" }));
app.use(express.static(path.join(__dirname, "public")));

const API_KEY = process.env.FAMWISE_API_KEY;
const BASE_URL = (process.env.FAMWISE_BASE_URL || "https://famwise.co").replace(/\/+$/, "");

function jsonError(res, status, message) {
  return res.status(status).json({ ok: false, error: message });
}

function validAmount(value) {
  const n = Number(value);
  return Number.isFinite(n) && n >= 1 && n <= 100000 && Math.round(n * 100) === n * 100;
}

// Demo in-memory order map. Replace with a persistent database before production.
// Each order is credited at most once in this process; production must use a DB transaction + unique order_id.
const orders = new Map();

app.post("/api/add-funds/create", async (req, res) => {
  if (!API_KEY) return jsonError(res, 503, "Payment gateway is not configured on the server.");

  const amount = Number(req.body?.amount);
  if (!validAmount(amount)) return jsonError(res, 400, "Enter a valid amount (minimum ₹1, up to 2 decimal places).");

  const createPath = process.env.FAMWISE_CREATE_PATH;
  const amountField = process.env.FAMWISE_AMOUNT_FIELD;
  if (!createPath || !amountField) {
    return jsonError(
      res,
      501,
      "Famwise order-creation endpoint/field mapping is not configured. Add the exact values from official Famwise API documentation."
    );
  }

  // Build the request only from documented field names. Never accept arbitrary URLs from the browser.
  const payload = { [amountField]: amount };
  if (process.env.FAMWISE_CURRENCY_FIELD) {
    payload[process.env.FAMWISE_CURRENCY_FIELD] = process.env.FAMWISE_CURRENCY_VALUE || "INR";
  }
  if (process.env.FAMWISE_CALLBACK_URL) {
    payload.callback_url = process.env.FAMWISE_CALLBACK_URL;
  }

  try {
    const url = new URL(createPath, BASE_URL);
    if (url.origin !== BASE_URL) return jsonError(res, 400, "Invalid configured gateway path.");

    const gatewayResponse = await fetch(url, {
      method: process.env.FAMWISE_CREATE_METHOD || "POST",
      headers: {
        "Authorization": `Bearer ${API_KEY}`,
        "Content-Type": "application/json",
        "Accept": "application/json"
      },
      body: JSON.stringify(payload)
    });

    const data = await gatewayResponse.json().catch(() => ({}));
    if (!gatewayResponse.ok) {
      return jsonError(res, 502, data?.error?.message || "Payment gateway could not create an order.");
    }

    // These response properties match the fields used in the Telegram code.
    if (!data.order_id || !data.upi_intent) {
      return jsonError(res, 502, "Gateway response is missing order_id or upi_intent. Check the official API response format.");
    }

    orders.set(String(data.order_id), {
      orderId: String(data.order_id),
      amount: Number(data.amount ?? amount),
      status: "pending",
      credited: false,
      createdAt: new Date().toISOString()
    });

    res.json({
      ok: true,
      orderId: String(data.order_id),
      amount: Number(data.amount ?? amount),
      upiIntent: String(data.upi_intent)
    });
  } catch {
    jsonError(res, 502, "Could not reach the payment gateway.");
  }
});

app.get("/api/add-funds/status/:orderId", async (req, res) => {
  if (!API_KEY) return jsonError(res, 503, "Payment gateway is not configured on the server.");

  const orderId = String(req.params.orderId || "");
  if (!/^[a-zA-Z0-9_-]{1,128}$/.test(orderId)) return jsonError(res, 400, "Invalid order ID.");

  try {
    const url = new URL(`/api/public/v1/payment-links/${encodeURIComponent(orderId)}`, BASE_URL);
    const gatewayResponse = await fetch(url, {
      headers: { "Authorization": `Bearer ${API_KEY}`, "Accept": "application/json" }
    });
    const data = await gatewayResponse.json().catch(() => ({}));
    if (!gatewayResponse.ok) return jsonError(res, 502, data?.error?.message || "Could not verify payment.");

    const order = orders.get(orderId);
    const status = String(data.status || "unknown").toLowerCase();

    // IMPORTANT: The starter does not credit a real wallet. A logged-in user ID,
    // database transaction, and unique order constraint must be added before production.
    if (order) order.status = status;

    res.json({
      ok: true,
      orderId,
      status,
      amount: Number(data.amount ?? order?.amount ?? 0),
      walletCreditApplied: false,
      message: status === "paid"
        ? "Payment verified. Connect this result to your authenticated wallet database before enabling credits."
        : status === "active" ? "Payment is still pending." : status === "expired" ? "This payment order expired." : "Payment status received."
    });
  } catch {
    jsonError(res, 502, "Could not reach the payment gateway.");
  }
});

app.get("/api/health", (_req, res) => res.json({ ok: true }));

const port = Number(process.env.PORT || 3000);
app.listen(port, () => console.log(`Add Funds starter running on http://localhost:${port}`));

# Famwise Add Funds — website starter

This is a secure **integration scaffold**, not a production-ready wallet or complete payment integration.

## What is included
- Add amount form (minimum ₹1)
- Server-side Famwise API key handling
- Payment-order creation adapter (must be configured from official docs)
- QR display from the `upi_intent` response
- Payment status check using the status URL shown in your Telegram code:
  `GET https://famwise.co/api/public/v1/payment-links/{order_id}`
- No Telegram commands or Telegram chat IDs

## Important: rotate the key you posted
The live API key shared in chat should be revoked/rotated in the Famwise dashboard. Put the **new** key only in `.env`. Never paste it into `public/` files or share it in screenshots.

## Run locally
1. Install Node.js 18 or newer.
2. Copy `.env.example` to `.env`.
3. Put the newly rotated key in `FAMWISE_API_KEY`.
4. Run `npm install`.
5. Run `npm start`.
6. Open `http://localhost:3000`.

## Required before order creation works
The Telegram code supplied includes the status-check endpoint and the response fields `order_id`, `upi_intent`, `amount`, but it does **not** include the create-payment request URL or request field names.

Read the official Famwise API documentation and set:
- `FAMWISE_CREATE_PATH`
- `FAMWISE_AMOUNT_FIELD`
- optionally `FAMWISE_CURRENCY_FIELD`
- optionally `FAMWISE_CALLBACK_URL`

Do not guess these values. The server deliberately returns a clear configuration error until they are supplied.

## Required before production wallet credits
This starter intentionally does **not** credit a wallet. To safely enable real credits, add:
- authenticated website users (never trust a user ID sent by the browser)
- a persistent database for orders and wallet balances
- a database transaction and unique constraint on `order_id` so one payment can only credit once
- server-side verification of the paid amount and currency
- verified webhook signature handling if Famwise supports webhooks; otherwise controlled server-side polling
- rate limiting, CSRF/origin protections, logging without secrets, and HTTPS

Do not treat a browser redirect or QR scan as proof of payment. Only a verified `paid` status from the gateway can trigger a wallet credit.
